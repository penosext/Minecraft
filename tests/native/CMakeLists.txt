cmake_minimum_required(VERSION 3.20)
project(minecraft_native_regressions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_SUPPRESS_REGENERATION ON)
set(BUILD_TESTING OFF CACHE BOOL "Standalone module tests" FORCE)
set(ORIGINAL_BIOMES_BUILD_TESTS OFF CACHE BOOL "Standalone biome tests" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${ROOT}/port/original-core" "${CMAKE_BINARY_DIR}/core")
add_subdirectory("${ROOT}/port/original-core/biomes" "${CMAKE_BINARY_DIR}/biomes")
add_subdirectory("${ROOT}/port/original-core/terrain" "${CMAKE_BINARY_DIR}/terrain")
add_subdirectory("${ROOT}/port/original-core/structures" "${CMAKE_BINARY_DIR}/structures")
add_subdirectory("${ROOT}/port/original-core/decoration" "${CMAKE_BINARY_DIR}/decoration")
add_subdirectory("${ROOT}/port/original-core/carvers" "${CMAKE_BINARY_DIR}/carvers")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(GENERATOR IN ITEMS generate-terrain.py generate-items.py generate-item-icons.py generate-recipes.py extract-containers.py)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/${GENERATOR}" RESULT_VARIABLE GENERATE_RESULT)
    if(NOT GENERATE_RESULT EQUAL 0)
        message(FATAL_ERROR "Catalog/placement generation failed: ${GENERATOR}")
    endif()
endforeach()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/extract-placement.py" --check RESULT_VARIABLE PLACEMENT_RESULT)
if(NOT PLACEMENT_RESULT EQUAL 0)
    message(FATAL_ERROR "Original placement rule audit failed")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/extract-collision.py" --check RESULT_VARIABLE COLLISION_RESULT)
if(NOT COLLISION_RESULT EQUAL 0)
    message(FATAL_ERROR "Original collision rule audit failed")
endif()
# Survival HUD sprites (tools/generate-hud.py); separate from the shared list above.
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tools/generate-hud.py" RESULT_VARIABLE HUD_RESULT)
if(NOT HUD_RESULT EQUAL 0)
    message(FATAL_ERROR "Survival HUD sprite generation failed")
endif()
set(NATIVE "${ROOT}/jsapi/src/MinecraftRenderer")
add_library(world_runtime STATIC "${NATIVE}/World.cpp" "${NATIVE}/OriginalWorld.cpp" "${NATIVE}/Scene.cpp" "${NATIVE}/Inventory.cpp" "${NATIVE}/ItemPlacement.cpp" "${NATIVE}/BlockCollision.cpp" "${NATIVE}/BlockInteraction.cpp" "${NATIVE}/Crafting.cpp" "${NATIVE}/Redstone.cpp" "${NATIVE}/BlockTicks.cpp" "${NATIVE}/WorldRules.cpp" "${NATIVE}/ToolRules.cpp" "${NATIVE}/GameActions.cpp" "${NATIVE}/PlayerSurvival.cpp" "${NATIVE}/BlockEntities.cpp" "${NATIVE}/Smelting.cpp" "${NATIVE}/MiningEffects.cpp" "${NATIVE}/WorldJob.cpp" "${NATIVE}/ChunkPump.cpp")
target_sources(world_runtime PRIVATE "${NATIVE}/SurvivalFrame.cpp")
target_include_directories(world_runtime PUBLIC "${NATIVE}" "${ROOT}/jsapi/build/generated")
target_link_libraries(world_runtime PUBLIC original_terrain original_structures original_decoration original_carvers original_biomes)
target_compile_options(world_runtime PRIVATE -Wall -Wextra -Werror=return-type)
add_library(renderer_runtime STATIC "${NATIVE}/WorldRenderer.cpp" "${NATIVE}/ActorRenderer.cpp" "${NATIVE}/MobRenderer.cpp" "${NATIVE}/LegacyBlock.cpp" "${NATIVE}/SpecialShapes.cpp" "${NATIVE}/InventoryRenderer.cpp")
target_link_libraries(renderer_runtime PUBLIC world_runtime)
target_sources(renderer_runtime PRIVATE "${NATIVE}/SurvivalHud.cpp")
target_compile_options(renderer_runtime PRIVATE -O3)
add_executable(device_save_probe "${ROOT}/tests/device-save-probe.cpp")
target_link_libraries(device_save_probe PRIVATE renderer_runtime)
target_compile_options(device_save_probe PRIVATE -UNDEBUG)
# v3 vs v4 (carvers/ores) generation + lighting timings; not a ctest.
add_executable(carvers_perf_probe "${ROOT}/tests/carvers-perf-probe.cpp")
target_link_libraries(carvers_perf_probe PRIVATE renderer_runtime)
enable_testing()
add_executable(test_structure-world-smoke "${ROOT}/tests/structure-world-smoke.cpp")
target_link_libraries(test_structure-world-smoke PRIVATE renderer_runtime)
target_compile_options(test_structure-world-smoke PRIVATE -UNDEBUG)
add_test(NAME structure-world-smoke COMMAND test_structure-world-smoke)
set_tests_properties(structure-world-smoke PROPERTIES TIMEOUT 180)
foreach(NAME IN ITEMS original-world scene combat mining world-interaction item-placement block-collision block-interaction crafting redstone block-ticks chunk-streaming preload-settings compact-window world-rules save-v2 world-job chunk-pump carvers-world)
    add_executable(test_${NAME} "${ROOT}/tests/${NAME}-test.cpp")
    target_link_libraries(test_${NAME} PRIVATE world_runtime)
    target_compile_options(test_${NAME} PRIVATE -UNDEBUG)
    add_test(NAME ${NAME} COMMAND test_${NAME})
    set_tests_properties(${NAME} PROPERTIES TIMEOUT 600)
endforeach()
foreach(NAME IN ITEMS survival survival-hud)
    add_executable(test_${NAME} "${ROOT}/tests/${NAME}-test.cpp")
    target_link_libraries(test_${NAME} PRIVATE renderer_runtime)
    target_compile_options(test_${NAME} PRIVATE -UNDEBUG)
    add_test(NAME ${NAME} COMMAND test_${NAME})
    set_tests_properties(${NAME} PROPERTIES TIMEOUT 300)
endforeach()
foreach(NAME IN ITEMS plant-placement containers actor-renderer mob-renderer world-renderer structure-renderer lighting-celestial lighting-local inventory item-state-inventory item-icons async-chunk world-scene-integration)
    if(NAME STREQUAL "world-scene-integration")
        set(SOURCE "${ROOT}/tests/${NAME}.cpp")
    else()
        set(SOURCE "${ROOT}/tests/${NAME}-test.cpp")
    endif()
    if(EXISTS "${SOURCE}")
        add_executable(test_${NAME} "${SOURCE}")
        target_link_libraries(test_${NAME} PRIVATE renderer_runtime)
        target_compile_options(test_${NAME} PRIVATE -UNDEBUG)
        add_test(NAME ${NAME} COMMAND test_${NAME})
        set_tests_properties(${NAME} PROPERTIES TIMEOUT 180)
    endif()
endforeach()
