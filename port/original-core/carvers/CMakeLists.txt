cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(minecraft_original_carvers VERSION 0.1 LANGUAGES CXX)
    file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}" CARVERS_SOURCE)
    file(REAL_PATH "${CMAKE_BINARY_DIR}" CARVERS_BINARY)
    cmake_path(IS_PREFIX CARVERS_SOURCE "${CARVERS_BINARY}" NORMALIZE INSIDE_CARVERS)
    if(NOT INSIDE_CARVERS OR CARVERS_SOURCE STREQUAL CARVERS_BINARY)
        message(FATAL_ERROR "Keep standalone outputs below original-core/carvers")
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/core")
endif()
if(NOT TARGET original_core)
    message(FATAL_ERROR "Add original-core before original_carvers")
endif()
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
set(CARVERS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
# Check only: generated sources are committed; configure never rewrites them.
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CARVERS_ROOT}/extract.py" --check COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CARVERS_ROOT}/extract.py" "${CARVERS_ROOT}/upstream-sha256.json")
add_custom_target(carvers_source_audit ALL
    COMMAND "${Python3_EXECUTABLE}" "${CARVERS_ROOT}/extract.py" --check VERBATIM)
add_library(original_carvers STATIC Carvers.cpp generated/Original.cpp)
add_dependencies(original_carvers carvers_source_audit)
target_include_directories(original_carvers PUBLIC "${CARVERS_ROOT}" PRIVATE "${CARVERS_ROOT}/generated")
target_link_libraries(original_carvers PUBLIC original_core)
target_compile_features(original_carvers PUBLIC cxx_std_11)
target_compile_options(original_carvers PRIVATE -fwrapv -ffp-contract=off -fno-fast-math -fsigned-char -UNDEBUG)
include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(original_carvers_tests tests.cpp)
    target_link_libraries(original_carvers_tests PRIVATE original_carvers Threads::Threads)
    target_compile_options(original_carvers_tests PRIVATE -UNDEBUG)
    foreach(CASE IN ITEMS caves determinism ores threaded)
        add_test(NAME "carvers_${CASE}" COMMAND original_carvers_tests "${CASE}")
        set_tests_properties("carvers_${CASE}" PROPERTIES TIMEOUT 180)
    endforeach()
    add_test(NAME carvers_source_bytes COMMAND "${Python3_EXECUTABLE}" "${CARVERS_ROOT}/extract.py" --check)
endif()
