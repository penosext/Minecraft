cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(minecraft_original_decoration VERSION 0.1 LANGUAGES CXX)
    file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}" DECORATION_SOURCE)
    file(REAL_PATH "${CMAKE_BINARY_DIR}" DECORATION_BINARY)
    cmake_path(IS_PREFIX DECORATION_SOURCE "${DECORATION_BINARY}" NORMALIZE INSIDE_DECORATION)
    if(NOT INSIDE_DECORATION OR DECORATION_SOURCE STREQUAL DECORATION_BINARY)
        message(FATAL_ERROR "Keep standalone outputs below original-core/decoration")
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/core")
endif()
if(NOT TARGET original_core)
    message(FATAL_ERROR "Add original-core before original_decoration")
endif()
find_package(Threads REQUIRED)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
set(DECORATION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${DECORATION_ROOT}/extract.py" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${DECORATION_ROOT}/extract.py" "${DECORATION_ROOT}/upstream-sha256.json")
add_custom_target(decoration_source_audit ALL
    COMMAND "${Python3_EXECUTABLE}" "${DECORATION_ROOT}/extract.py" --check VERBATIM)
add_library(original_decoration STATIC Decoration.cpp Platform.cpp generated/Original.cpp)
add_dependencies(original_decoration decoration_source_audit)
target_include_directories(original_decoration PUBLIC "${DECORATION_ROOT}" PRIVATE "${DECORATION_ROOT}/generated")
target_link_libraries(original_decoration PUBLIC original_core Threads::Threads)
target_compile_features(original_decoration PUBLIC cxx_std_11)
target_compile_options(original_decoration PRIVATE -fwrapv -ffp-contract=off -fno-fast-math -fsigned-char -UNDEBUG)
include(CTest)
if(BUILD_TESTING)
    add_executable(original_decoration_tests tests.cpp)
    target_link_libraries(original_decoration_tests PRIVATE original_decoration)
    target_include_directories(original_decoration_tests PRIVATE "${DECORATION_ROOT}/generated")
    target_compile_options(original_decoration_tests PRIVATE -fsigned-char -UNDEBUG)
    foreach(CASE IN ITEMS rawfeature metadata schedule determinism seams desert bounds container plants threaded)
        add_test(NAME "decoration_${CASE}" COMMAND original_decoration_tests "${CASE}")
        set_tests_properties("decoration_${CASE}" PROPERTIES TIMEOUT 180)
    endforeach()
    add_test(NAME decoration_source_bytes COMMAND "${Python3_EXECUTABLE}" "${DECORATION_ROOT}/extract.py" --check)
    option(ORIGINAL_DECORATION_REAL_SOURCE_TESTS "Test genuine Source -> terrain -> decoration bridge" ON)
    if(ORIGINAL_DECORATION_REAL_SOURCE_TESTS)
        if(NOT TARGET original_biomes)
            set(ORIGINAL_BIOMES_BUILD_TESTS OFF CACHE BOOL "Build biome tests" FORCE)
            add_subdirectory("${DECORATION_ROOT}/../biomes" "${CMAKE_CURRENT_BINARY_DIR}/test-biomes")
        endif()
        if(TARGET original_terrain)
            set(DECORATION_TEST_TERRAIN original_terrain)
        else()
            # Read and compile existing audited terrain sources only. Do not run
            # its configure-time extractor (which writes outside our directory).
            set(TEST_TERRAIN_ROOT "${DECORATION_ROOT}/../terrain")
            execute_process(COMMAND "${Python3_EXECUTABLE}" "${TEST_TERRAIN_ROOT}/extract.py"
                --output "${TEST_TERRAIN_ROOT}/generated" --check COMMAND_ERROR_IS_FATAL ANY)
            add_library(decoration_test_terrain STATIC "${TEST_TERRAIN_ROOT}/Adapter.cpp"
                "${TEST_TERRAIN_ROOT}/generated/TerrainOriginal.cpp")
            target_include_directories(decoration_test_terrain PUBLIC "${TEST_TERRAIN_ROOT}"
                PRIVATE "${TEST_TERRAIN_ROOT}/generated")
            target_compile_definitions(decoration_test_terrain PRIVATE _LARGE_WORLDS=1)
            target_link_libraries(decoration_test_terrain PUBLIC original_core Threads::Threads)
            set(DECORATION_TEST_TERRAIN decoration_test_terrain)
        endif()
        add_executable(original_decoration_real_tests real_source_tests.cpp)
        target_link_libraries(original_decoration_real_tests PRIVATE original_decoration original_biomes ${DECORATION_TEST_TERRAIN})
        target_compile_options(original_decoration_real_tests PRIVATE -UNDEBUG)
        add_test(NAME decoration_real_source COMMAND original_decoration_real_tests)
        set_tests_properties(decoration_real_source PROPERTIES TIMEOUT 180)
    endif()
endif()
