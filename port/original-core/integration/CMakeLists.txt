cmake_minimum_required(VERSION 3.20)
project(minecraft_original_generation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Each component audits generated source bytes on every build.
set(CMAKE_SUPPRESS_REGENERATION ON)
include(CTest)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_BINARY_DIR}/noise")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../biomes" "${CMAKE_BINARY_DIR}/biomes")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../terrain" "${CMAKE_BINARY_DIR}/terrain")
add_executable(original_generation_tests generation_tests.cpp)
target_link_libraries(original_generation_tests PRIVATE original_core original_biomes original_terrain)
target_compile_options(original_generation_tests PRIVATE -fwrapv -ffp-contract=off -fno-fast-math -UNDEBUG)
add_test(NAME original_generation_chain COMMAND original_generation_tests)
set_tests_properties(original_generation_chain PROPERTIES TIMEOUT 120)
